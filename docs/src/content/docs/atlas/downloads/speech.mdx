import DownloadList from '../../../../components/DownloadList.astro';
import speech from '../../../../data/speech.json';

export const rows = (section) => speech.sections[section].map(([detail, id]) => {
	const { url, ...model } = speech.models[id];
	return { ...model, detail, href: url };
});
